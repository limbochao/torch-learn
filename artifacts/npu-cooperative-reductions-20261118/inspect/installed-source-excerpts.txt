411:     @staticmethod
412:     def should_use_persistent_reduction(
413:         features: SIMDKernelFeatures, cooperative_reduction: bool
414:     ) -> bool:
415:         """
416:         Heuristic to decide if a persistent reduction should be used.
417:         """
418:         if not config.triton.persistent_reductions:
419:             return False
420:         threshold = {
421:             ReductionHint.INNER: 1024,
422:         }.get(features.get_reduction_hint(), 64)
423: 
424:         if features.get_reduction_hint() not in (
425:             ReductionHint.INNER,
426:             ReductionHint.OUTER_TINY,
427:         ):
428:             bounds = bound_sympy(features.reduction_numel)
429:             lower = bounds.lower
430:             upper = bounds.upper
431: 
432:             if not all(
433:                 (
434:                     (isinstance(bound, int) or bound.is_constant())
435:                     and not torch.utils._sympy.numbers.is_infinite(bound)
436:                 )
437:                 for bound in (lower, upper)
438:             ):
439:                 return False
440: 
441:             lower = next_power_of_2(int(lower))
442:             upper = next_power_of_2(int(upper))
443: 
444:             # If we are coalescing on xblock (not ReductionHint.INNER) and this is not a tiny kernel
445:             # (not ReductionHint.OUTER_TINY), do not use persistent reduction if it induces tile
446:             # quantization. Persistent reduction forces rblock == rnumel, if the bounds between lower
447:             # and upper are large, for the lower values we will be masking off large % of read/writes,
448:             # when we could expand the coalescing xblock instead.
449:             if lower != upper:
450:                 return False
451: 
452:         if cooperative_reduction:
453:             # The RSPLIT of cooperative reductions means each thread block is operating on fewer elements
454:             # The default fallback will be used if optimizations hint is not provided. The default fallback
455:             # is >> 32.
456:             threshold *= 32 // min(
457:                 V.graph.sizevars.optimization_hint(features.numel), 32
458:             )
459: 
460:         # If multi_kernel is enabled, we do more aggressive persistent reduction.
461:         # This may result in some persistent reductions slower than the
462:         # corresponding non-persistent reductions. MultiKernel will do benchmarking
463:         # to pick the faster one.
464:         if config.triton.multi_kernel:
465:             threshold *= 16
466: 
467:         return V.graph.sizevars.statically_known_leq(
468:             features.reduction_numel, threshold

131: def welford(mean, m2, weight, dim):
132:     """Override upstream welford to avoid tl.reduce with 3 source operands.
133: 
134:     NPU's Triton compiler (TritonOpConverter.cpp) asserts srcs.size() <= 2,
135:     but upstream welford uses tl.reduce((mean,m2,weight), dim, welford_combine)
136:     with 3 operands. This version decomposes into 3 single-operand reduces.
137:     """
138:     sum_x = weight * mean
139:     sum_x2 = m2 + weight * mean * mean
140:     total_sum_x = tl.reduce(sum_x, dim, _welford_sum_combine)
141:     total_sum_x2 = tl.reduce(sum_x2, dim, _welford_sum_combine)
142:     total_weight = tl.reduce(weight, dim, _welford_sum_combine)
143:     new_mean = tl.where(total_weight == 0.0, 0.0, total_sum_x / total_weight)
144:     new_m2 = total_sum_x2 - total_weight * new_mean * new_mean
145:     return new_mean, new_m2, total_weight

379: @builtin
380: def fixpipe(
381:     src: tl.tensor,
382:     dst: Union[None, bl.buffer] = None,
383:     dma_mode: FixpipeDMAMode = FixpipeDMAMode.NZ2ND,
384:     dual_dst_mode: FixpipeDualDstMode = FixpipeDualDstMode.NO_DUAL,
385:     _semantic=None,
386: ) -> Union[None, tl.tensor]:
387:     """
388:     Directly store a tensor on L0C to a local buffer via fixpipe.
389:     Fixpipe is pipeline that performing data movement from L0C to other memory hierarchies.
390:     Currently support:
391:         - L0C to UB (for Ascend910_95 sereies)
392: 
393:     :param src: the source tensor, Must be located in the l0C memory region.
394:     :type src: tl.tensor
395:     :param dst: The destination buffer in UB. If None, an empty tensor will be created.
396:                 Must be buffer type in UB if provided.
397:     :type dst: bl.buffer | None
398:     :param dma_mode: DMA transfer mode, "nz2nd" enables NZ to ND layout transformation
399:     :type dma_mode: FixpipeDMAMode
400:     :param dual_dst_mode: Dual destination mode for split operations
401:     :type dual_dst_mode: FixpipeDualDstMode
402:     :return: If dst is None, returns the created tensor; otherwise returns None.
403:     :rtype: tl.tensor | None
404:     """
405:     if not is_register_based_arch(_semantic.builder.options.arch):
406:         raise RuntimeError("this feature is only supported on Ascend910_95 and Ascend960")
407:     if not isinstance(src, tl.tensor):
408:         raise TypeError("src is not of tensor type")
409: 
410:     if dst is not None:
411:         if not isinstance(dst, bl.buffer):
412:             raise TypeError("dst must be buffer type or None")
413:         if dst.space != ascend_address_space.UB:
414:             raise TypeError("dst's AddressSpace must be UB")
415:         if len(dst.shape) == 2 and (dst.type.element_ty == tl.float32 or dst.type.element_ty == tl.int32):
416:             N = dst.shape[1]
417:             if N % 8 != 0:
418:                 raise ValueError("32b Fixpipe last dim must be aligned to 8")
419:             if (dma_mode != FixpipeDMAMode.NZ2ND) and (N % 16 != 0):
420:                 raise ValueError("32b non-NZ2ND Fixpipe last dim must be aligned to 16")
421:             if (dual_dst_mode == FixpipeDualDstMode.COLUMN_SPLIT) and (N % 32 != 0):
422:                 raise ValueError("32b Column split dual Fixpipe last dim must be aligned to 32")
423:             M = dst.shape[0]
424:             if (dma_mode == FixpipeDMAMode.NZ2DN) and (M % 8 != 0):
425:                 raise ValueError("32b NZ2DN Fixpipe first dim must be aligned to 8")
426:         dst16bits = (dst.type.element_ty == tl.float16 or dst.type.element_ty == tl.int16
427:                      or dst.type.element_ty == tl.bfloat16)
